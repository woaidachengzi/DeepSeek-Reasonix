def launch(app_path,identifier,root,phase,configuration):
 if phase=='explicit':
  assert not profile.package.matching_package_is_running(identifier),'Preview must be stopped before backup restore'
  core=root/'home/Library/Application Support'/identifier/'reasonix-core'
  assert not profile.package.own_sidecars(root/'tmp',app/'Contents/MacOS/reasonix-desktop-bridge')
  backups=list((core/'backups').glob('*/config.toml'));assert len(backups)==1

  restored=restored_home/'.reasonix';restored.mkdir(mode=0o700)

  env={k:os.environ[k] for k in ('PATH','USER','LOGNAME','LANG') if k in os.environ}

  assert receipt['restoredTreeUnchanged'] and receipt['previewAndBackupTreeUnchanged']
  receipts.append(receipt)
 return original_launch(app_path,identifier,root,phase,configuration)
